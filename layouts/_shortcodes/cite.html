{{- /* Inline citation shortcode. Usage: cite "KEY" Renders "Surname Year" (2 authors: "A and B", 3+ authors: "A et al."), linked to the matching entry in the bibliography. Editors are used when there is no author; entries with neither fall back to the title and finally to the citation key. Key presence is tested with ``isset``, so an entry that exists but carries no fields (e.g. a note-only entry after a partial regeneration) is still a known key and does not warn. Only genuinely absent keys render "[key]" and emit a warning; a call without an argument renders "[cite]" and warns instead of aborting the build, and a missing data file degrades to "[key]" after a warning. The link target is "#ref-KEY" on the current page. Pages without a {{< bibliography >}} shortcode link to the first page that has one instead, so citations always resolve. Data: data/refs.json (generated by scripts/bib_to_json.py from refs.bib). */ -}} {{- $key := .Get 0 -}} {{- $refs := hugo.Data.refs -}} {{- $found := false -}} {{- $entry := dict -}} {{- $anchor := $key -}} {{- $label := "" -}} {{- if not $refs -}} {{- warnf "cite: hugo.Data.refs is missing (key %q on %s)" (default "" $key) .Page.RelPermalink -}} {{- else if not $key -}} {{- warnf "cite: no citation key given on %s" .Page.RelPermalink -}} {{- else -}} {{- if isset $refs $key -}} {{- $entry = index $refs $key -}} {{- $found = true -}} {{- else if isset $refs (lower $key) -}} {{- /* Hugo may normalise data keys to lower case; try that before failing. */ -}} {{- $anchor = lower $key -}} {{- $entry = index $refs $anchor -}} {{- $found = true -}} {{- else -}} {{- warnf "cite: unknown citation key %q on %s" $key .Page.RelPermalink -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- if $key -}}[{{ $key }}]{{- else -}}[cite]{{- end -}} {{- else -}} {{- $names := slice -}} {{- with $entry.author -}} {{- $names = split . " and " -}} {{- else -}} {{- with $entry.editor -}} {{- $names = split . " and " -}} {{- end -}} {{- end -}} {{- $surnames := slice -}} {{- range $names -}} {{- $name := trim . " " -}} {{- if $name -}} {{- if in $name "," -}} {{- $surnames = $surnames | append (trim (index (split $name ",") 0) " ") -}} {{- else -}} {{- $parts := split $name " " -}} {{- $surnames = $surnames | append (index $parts (sub (len $parts) 1)) -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq (len $surnames) 1 -}} {{- $label = index $surnames 0 -}} {{- else if eq (len $surnames) 2 -}} {{- $label = printf "%s and %s" (index $surnames 0) (index $surnames 1) -}} {{- else if gt (len $surnames) 2 -}} {{- $label = printf "%s et al." (index $surnames 0) -}} {{- else -}} {{- $label = $entry.title | default $key -}} {{- end -}} {{- with $entry.year -}} {{- $label = printf "%s %s" $label . -}} {{- end -}} {{- /* Link to the bibliography on this page; if the page has none, use the first page that does (the bibliography shortcode is used only once). */ -}} {{- $href := printf "#ref-%s" $anchor -}} {{- if not (.Page.HasShortcode "bibliography") -}} {{- $bibPage := "" -}} {{- range site.RegularPages -}} {{- if and (not $bibPage) (.HasShortcode "bibliography") -}} {{- $bibPage = .RelPermalink -}} {{- end -}} {{- end -}} {{- with $bibPage -}} {{- $href = printf "%s#ref-%s" . $anchor -}} {{- end -}} {{- end -}} {{ $label }} {{- end -}}
